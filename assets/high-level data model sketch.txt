Project
  ├─ id (PK)
  ├─ name
  ├─ created_at_utc
  ├─ archived_at_utc  (nullable)
  └─ total_planned_minutes (cached, optional)

Task
  ├─ id (PK)
  ├─ project_id  (FK→Project)
  ├─ name
  ├─ estimated_minutes
  └─ archived_at_utc (nullable)   // when project completes, cascade set

Block  // “1‑hour slot”—may hold ≥1 Task
  ├─ id (PK)
  ├─ task_id   (FK→Task)   // or nullable if “unspecified”?
  ├─ day_local  (DATE)     // 2025‑07‑27
  ├─ start_minute_of_day   // e.g. 8 × 60 + 35 = 515
  ├─ length_minutes        // default 60
  └─ position_in_block     // 0 or 1 when you have 2 tasks in same block

TaskCompletion
  ├─ id (PK)
  ├─ block_id (FK→Block)
  ├─ minutes_completed
  └─ completed_at (DATETIME)
